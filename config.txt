# Настройки звездного небосвода и созвездий / CmdStars Configuration
#
# 1. Стартовая цветовая тема / Startup color theme:
#    Допустимые значения: от 1 до 5 / Values from 1 to 5
#    1 - Classic (Белые звезды, серые созвездия, черный фон)
#    2 - Deep Sky (Голубоватые звезды, синие созвездия, темно-синий фон)
#    3 - Nebula (Красновато-оранжевые звезды, линии пурпурные, фиолетовый фон)
#    4 - Gold (Теплые золотые звезды и линии, темно-коричневый фон)
#    5 - Matrix (Зеленые звезды и линии, черный фон)
theme = 1
#
# 2. Режим рендеринга звездного неба / Star rendering mode:
#    Допустимые значения: classic или realistic (по умолчанию: classic)
#    classic - звезды рисуются звездочками и точками (★, •, *, ·)
#    realistic - звезды рисуются затеняющими блоками (█, ▓, ▒, ░) с ореолами свечения и Млечным Путем
render_mode = classic
#
# 3. Множитель скорости вращения неба / Sky rotation speed multiplier:
#    Значения от -5.0 до 5.0 (по умолчанию: 1.0) / Rotation speed multiplier
rotation_speed = 1.0
#
# 4. Скорость мерцания звезд / Star twinkling speed:
#    Значения от 0.1 до 5.0 (по умолчанию: 1.5) / Twinkle frequency
twinkle_speed = 1.5
#
# 5. Отображать созвездия по умолчанию / Show all constellations on boot:
#    Допустимые значения: true или false (по умолчанию: true)
show_all = true
#
# 6. Отображать зодиакальный пояс по умолчанию / Show zodiac ecliptic line on boot:
#    Допустимые значения: true или false (по умолчанию: true)
show_zodiac = true
#
# 7. Количество фоновых звезд / Background stars count:
#    Количество свободно мерцающих звезд (по умолчанию: 250) / Star count limit
star_count = 250
#
# 8. Соотношение сторон шрифта / Font aspect ratio:
#    Коэффициент ширины/высоты (по умолчанию: 2.0). Регулирует округлость вращения.
aspect = 2.0
